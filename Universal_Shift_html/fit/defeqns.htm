FDCPE_Q0: FDCPE port map (Q(0),Q_D(0),clk,reset,'0','1');
     Q_D(0) <= ((S1 AND S0 AND Parallel_in(0))
      OR (S1 AND NOT S0 AND Serial_in)
      OR (NOT S1 AND S0 AND Q(1))
      OR (NOT S1 AND NOT S0 AND Q(0)));
FDCPE_Q1: FDCPE port map (Q(1),Q_D(1),clk,reset,'0','1');

      OR (S1 AND NOT S0 AND Serial_out)
      OR (NOT S1 AND S0 AND Q(2))
      OR (NOT S1 AND NOT S0 AND Q(1)));
FDCPE_Q2: FDCPE port map (Q(2),Q_D(2),clk,reset,'0','1');

      OR (S1 AND NOT S0 AND Q(1))

FDCPE_Q3: FDCPE port map (Q(3),Q_D(3),clk,reset,'0','1');

      OR (S1 AND NOT S0 AND Q(2))

FDCPE_Serial_out: FDCPE port map (Serial_out,Serial_out_D,clk,reset,'0','1');

      OR (S1 AND NOT S0 AND Serial_in)
      OR (NOT S1 AND S0 AND Q(1))
      OR (NOT S1 AND NOT S0 AND Serial_out));
